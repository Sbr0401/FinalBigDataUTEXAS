\sectionbox{DATASET}{
  \textbf{Source:} CC GENERAL (Kaggle)\\
  \textbf{Records:} 8{,}950 credit card customers\\
  \textbf{Variables:} 18 behavioral \& financial\\
  \textbf{After cleaning:} 8{,}636 records ($<$4\% removed)\\[2pt]
  All major financial variables are \textbf{right-skewed}
  (PURCHASES skew = 8.14; CASH\_ADVANCE = 5.17;
  BALANCE = 2.39). A log-transform was applied to BALANCE
  before the $t$-test to meet normality requirements.
}

\sectionbox{RESEARCH QUESTION}{
  \textit{``Do distinct patterns of credit card usage carry
  statistically measurable differences, and can we identify
  natural customer profiles through clustering?''}\\[2pt]
  Answered in two parts: \textbf{(1)} Formal hypothesis testing
  and \textbf{(2)} K-Means unsupervised clustering.
}

\sectionbox{H1 --- CONFIRMED \checkmark}{
  \textbf{Claim:} Heavy cash-advance users carry higher balances.\\[2pt]
  \begin{tabular}{@{}ll@{}}
    Test:       & Welch's two-sample $t$-test (one-tailed) \\
    $t$-stat:   & $\approx 28$ \\
    $df$:       & $\approx 2{,}889$ \\
    $p$-value:  & $< 2.2\times10^{-16}$ \\
    Decision:   & \textbf{Reject $H_0$} \\
  \end{tabular}\\[3pt]
  \result{Conclusion:}{Heavy CA users carry \textbf{significantly
  higher} balances. \texttt{CASH\_ADVANCE\_FREQUENCY} is a
  reliable credit-risk signal.}
}

\sectionbox{H2 --- DEBUNKED \texttimes}{
  \textbf{Claim:} Balance strongly predicts credit limit.\\[2pt]
  \begin{tabular}{@{}ll@{}}
    Test:       & Pearson correlation \texttt{cor.test()} \\
    $r$:        & $\approx 0.54$ \\
    $r^2$:      & $\approx 29\%$ variance explained \\
    $p$-value:  & $< 2.2\times10^{-16}$ \\
    Decision:   & \textbf{Debunked} \\
  \end{tabular}\\[3pt]
  \result{Key lesson:}{$p < 0.05 \neq$ practical strength.
  Balance explains only 29\% of credit-limit variance.
  Banks set limits \textit{prospectively} (income, credit
  history), not based on current debt.}
}

\columnbreak

\sectionbox{K-MEANS CLUSTERING (K = 5)}{
  \textbf{Input:} 8 standardized financial variables\\
  \textbf{Method:} K-Means, \texttt{nstart=25}, \texttt{set.seed(123)}\\
  \textbf{K selected by:} Elbow method + interpretability\\[4pt]

  \begin{tabular}{@{}p{3.6cm}@{\;}c@{}}
    \toprule
    \textbf{Cluster} & \textbf{\%} \\
    \midrule
    Low-Activity Customers  & 34\% \\
    Regular Purchasers      & 26\% \\
    Full-Payment Customers  & 18\% \\
    High CA Users           & 20\% \\
    High-Value Customers    & 1.3\% \\
    \bottomrule
  \end{tabular}\\[4pt]

  \textbf{High CA Users} cluster independently corroborates H1:
  highest CASH\_ADVANCE\_FREQUENCY $\rightarrow$ highest
  average BALANCE.
}

\sectionbox{KEY CORRELATIONS}{
  \begin{tabular}{@{}p{4.1cm}@{\;}c@{}}
    \toprule
    \textbf{Pair} & $r$ \\
    \midrule
    PURCHASES $\leftrightarrow$ PAYMENTS   & $+0.60$ \\

    PRCFULLPAYMENT $\leftrightarrow$ BALANCE & $-0.32$ \\

\sectionbox{TOOLS \& METHODS}{
  \begin{tabular}{@{}ll@{}}
    Language:   & R \\
    Wrangling:  & \texttt{dplyr} (\texttt{filter}, \texttt{mutate}) \\
    H1 test:    & \texttt{t.test(var.equal=FALSE)} \\
    H2 test:    & \texttt{cor.test(method="pearson")} \\
    Clustering: & \texttt{kmeans(centers=5, nstart=25)} \\

\sectionbox{OVERALL CONCLUSIONS}{
  \begin{itemize}
    \item \textbf{H1 Confirmed:} Cash-advance frequency is a
      measurable credit-risk predictor.
    \item \textbf{H2 Debunked:} Statistical significance alone
      is misleading with large $n$; always report $r^2$.
    \item \textbf{Clustering:} 5 interpretable profiles reveal
      distinct behavioral segments; exploratory, not definitive.
    \item \textbf{Core lesson:} Significance $\neq$ Practical
      relevance. Effect size matters.
  \end{itemize}
}
